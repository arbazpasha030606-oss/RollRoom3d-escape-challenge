	function createButton(x, y, width, height, text, callback) {
		var button = new Container();
		var bg = LK.getAsset('menuButton', {
			anchorX: 0.5,
			anchorY: 0.5
		});
		button.addChild(bg);
		var label = new Text2(text, {
			size: 60,
			fill: 0xFFFFFF
		});
		label.anchor.set(0.5, 0.5);
		button.addChild(label);
		button.x = x;
		button.y = y;
		button.width = width;
		button.height = height;
		button.interactive = true;
		button.callback = callback;
		button.down = function () {
			tween(label, {
				scale: 0.9
			}, {
				duration: 100
			});
		};
		button.up = function () {
			tween(label, {
				scale: 1
			}, {
				duration: 100
			});
			if (callback) {
				callback();
			}
		};
		self.addChild(button);
		menuButtons.push(button);
		return button;
	}

// Show level complete screen with timer and options
function showLevelCompleteScreen() {
	var timeElapsed = Math.floor((Date.now() - levelStartTime) / 1000);
	var countdownTime = 10;
	var levelCompleteContainer = new Container();
	// Semi-transparent background
	var bgRect = LK.getAsset('platform', {
		anchorX: 0.5,
		anchorY: 0.5
	});
	bgRect.width = 2048;
	bgRect.height = 2732;
	bgRect.alpha = 0.7;
	bgRect.tint = 0x000000;
	levelCompleteContainer.addChild(bgRect);
	// Title
	var titleText = new Text2('Level Complete!', {
		size: 120,
		fill: 0x00FF00
	});
	titleText.anchor.set(0.5, 0.5);
	titleText.x = 1024;
	titleText.y = 600;
	levelCompleteContainer.addChild(titleText);
	// Stats
	var statsText = new Text2('Level: ' + currentLevelData.number + '\nCoins Collected: ' + levelCoinsCollected + '\nTime: ' + timeElapsed + 's', {
		size: 80,
		fill: 0xFFFFFF
	});
	statsText.anchor.set(0.5, 0.5);
	statsText.x = 1024;
	statsText.y = 900;
	levelCompleteContainer.addChild(statsText);
	// Countdown timer
	var timerText = new Text2('Next Level in: ' + countdownTime + 's', {
		size: 70,
		fill: 0xFFD93D
	});
	timerText.anchor.set(0.5, 0.5);
	timerText.x = 1024;
	timerText.y = 1300;
	levelCompleteContainer.addChild(timerText);
	game.addChild(levelCompleteContainer);
	// Save level completion data
	if (!storage.levelProgress) {
		storage.levelProgress = {};
	}
	storage.levelProgress['level_' + currentLevelData.number] = {
		completed: true,
		coinsCollected: levelCoinsCollected,
		timeTaken: timeElapsed
	};
	// Countdown timer logic
	var countdownInterval = LK.setInterval(function () {
		countdownTime--;
		timerText.setText('Next Level in: ' + Math.max(0, countdownTime) + 's');
		if (countdownTime <= 0) {
			LK.clearInterval(countdownInterval);
			// Auto-advance to next level
			if (currentLevelData.number < 25) {
				cleanupLevel();
				startLevel(currentLevelData.number + 1);
			} else {
				cleanupLevel();
				gameState = 'menu';
				homeScreen = game.addChild(new HomeScreen());
			}
		}
	}, 1000);
	// Next Level button
	var nextLevelButton = new Container();
	var nextBg = LK.getAsset('menuButton', {
		anchorX: 0.5,
		anchorY: 0.5
	});
	nextLevelButton.addChild(nextBg);
	var nextLabel = new Text2('Next Level', {
		size: 60,
		fill: 0xFFFFFF
	});
	nextLabel.anchor.set(0.5, 0.5);
	nextLevelButton.addChild(nextLabel);
	nextLevelButton.x = 700;
	nextLevelButton.y = 1800;
	nextLevelButton.interactive = true;
	nextLevelButton.down = function () {
		tween(nextLabel, {
			scale: 0.9
		}, {
			duration: 100
		});
	};
	nextLevelButton.up = function () {
		tween(nextLabel, {
			scale: 1
		}, {
			duration: 100
		});
		LK.clearInterval(countdownInterval);
		if (currentLevelData.number < 25) {
			cleanupLevel();
			startLevel(currentLevelData.number + 1);
		}
	};
	levelCompleteContainer.addChild(nextLevelButton);
	// Back to Menu button
	var backButton = new Container();
	var backBg = LK.getAsset('menuButton', {
		anchorX: 0.5,
		anchorY: 0.5
	});
	backButton.addChild(backBg);
	var backLabel = new Text2('Back to Menu', {
		size: 60,
		fill: 0xFFFFFF
	});
	backLabel.anchor.set(0.5, 0.5);
	backButton.addChild(backLabel);
	backButton.x = 1348;
	backButton.y = 1800;
	backButton.interactive = true;
	backButton.down = function () {
		tween(backLabel, {
			scale: 0.9
		}, {
			duration: 100
		});
	};
	backButton.up = function () {
		tween(backLabel, {
			scale: 1
		}, {
			duration: 100
		});
		LK.clearInterval(countdownInterval);
		cleanupLevel();
		gameState = 'menu';
		homeScreen = game.addChild(new HomeScreen());
	};
	levelCompleteContainer.addChild(backButton);
}
// Cleanup level assets
function cleanupLevel() {
	gameState = 'menu';
	gameActive = false;
	// Remove GUI elements
	var guiChildren = LK.gui.top.children;
	for (var i = guiChildren.length - 1; i >= 0; i--) {
		if (guiChildren[i].name === 'coinDisplay' || guiChildren[i].name === 'levelDisplay') {
			LK.gui.top.removeChild(guiChildren[i]);
		}
	}
	// Remove level objects
	player = null;
	collectibles = [];
	obstacles = [];
	platforms = [];
	finishLine = null;
	// Clear game children
	for (var i = game.children.length - 1; i >= 0; i--) {
		game.removeChild(game.children[i]);
	}
}
// Generate level difficulty based on level number
function getLevelDifficulty(levelNum) {
	var baseSpeed = 2;
	var speedMultiplier = 1 + levelNum * 0.15;
	var difficulty = {
		number: levelNum,
		coinsRequired: Math.ceil(levelNum / 5),
		obstacleSpeed: baseSpeed + speedMultiplier,
		platformWidth: Math.max(100, 400 - levelNum * 8),
		diamondCoinChance: Math.min(0.5, levelNum * 0.03),
		minCompletionTime: 30 + levelNum * 5
	};
	if (levelNum <= 5) {
		difficulty.type = 'Easy';
	} else if (levelNum <= 10) {
		difficulty.type = 'Medium';
	} else if (levelNum <= 15) {
		difficulty.type = 'Hard';
	} else if (levelNum <= 20) {
		difficulty.type = 'Expert';
	} else {
		difficulty.type = 'Legendary';
	}
	return difficulty;
}
// Initialize gameplay level
function startLevel(levelNum) {
	gameState = 'gameplay';
	storage.currentLevel = levelNum;
	levelCoinsCollected = 0;
	levelStartTime = Date.now();
	gameActive = true;
	currentLevelData = getLevelDifficulty(levelNum);
	collectibles = [];
	obstacles = [];
	platforms = [];
	// Clear home screen and add game elements
	homeScreen.parent.removeChild(homeScreen);
	// Create player
	player = game.addChild(new Player());
	player.x = 1024;
	player.y = 2300;
	player.lastX = player.x;
	player.lastY = player.y;
	// Create platforms
	var platformY = 2200;
	var platformCount = 8 + currentLevelData.number;
	for (var p = 0; p < platformCount; p++) {
		var platform = game.addChild(new Platform());
		platform.x = 1024 + (Math.random() - 0.5) * 300;
		platform.y = platformY - p * 300;
		platforms.push(platform);
	}
	// Create finish line far below
	finishLine = game.addChild(new FinishLine());
	finishLine.x = 1024;
	finishLine.y = platformY - platformCount * 300 - 500;
	finishLine.lastWasIntersecting = false;
	// Create coins throughout level - spread across entire level length
	var coinCount = Math.floor(15 + currentLevelData.number * 0.5);
	for (var c = 0; c < coinCount; c++) {
		var coin = game.addChild(new Collectible());
		coin.x = 200 + Math.random() * 1648;
		coin.y = 2000 - c * 180 - Math.random() * 150;
		coin.lastY = coin.y;
		coin.value = 1;
		collectibles.push(coin);
	}
	// Add diamond coins - multiple per level for higher difficulties
	var diamondCount = Math.floor(currentLevelData.diamondCoinChance * 3);
	for (var d = 0; d < diamondCount; d++) {
		if (Math.random() < 0.4) {
			var diamond = game.addChild(new DiamondCoin());
			diamond.x = 300 + Math.random() * 1448;
			diamond.y = 1500 - d * 400 - Math.random() * 300;
			diamond.lastY = diamond.y;
			collectibles.push(diamond);
		}
	}
	// Create obstacles - increase count significantly for minimum 30-60 second completion
	var obstacleCount = Math.floor(8 + currentLevelData.number * 0.8);
	for (var o = 0; o < obstacleCount; o++) {
		var obstacle = game.addChild(new Obstacle());
		obstacle.x = 100 + Math.random() * 1848;
		obstacle.y = 2000 - o * 250;
		obstacle.speed = currentLevelData.obstacleSpeed * (0.8 + Math.random() * 0.4);
		obstacle.lastY = obstacle.y;
		obstacles.push(obstacle);
	}
	// Create UI for gameplay
	var coinDisplay = new Text2('Coins: ' + levelCoinsCollected, {
		size: 80,
		fill: 0xffd93d
	});
	coinDisplay.anchor.set(0, 0);
	coinDisplay.x = 50;
	coinDisplay.y = 50;
	coinDisplay.name = 'coinDisplay';
	LK.gui.top.addChild(coinDisplay);
	var levelDisplay = new Text2('Level: ' + currentLevelData.number, {
		size: 80,
		fill: 0xFFFFFF
	});
	levelDisplay.anchor.set(1, 0);
	levelDisplay.x = 2048 - 50;
	levelDisplay.y = 50;
	levelDisplay.name = 'levelDisplay';
	LK.gui.top.addChild(levelDisplay);
}
// Update game loop
game.update = function () {
	if (gameState === 'menu') {
		// Menu is handled by HomeScreen class
		return;
	}
	if (!gameActive || !player) {
		return;
	}
	// Update player (manual movement based on touch)
	player.lastX = player.x;
	player.lastY = player.y;
	// Update collectibles
	for (var c = collectibles.length - 1; c >= 0; c--) {
		var collectible = collectibles[c];
		collectible.lastY = collectible.y;
		collectible.update();
		// Check if off-screen
		if (collectible.y > 2732 || collectible.y < -100) {
			collectible.destroy();
			collectibles.splice(c, 1);
			continue;
		}
		// Check collision with player
		if (!collectible.collected && player.intersects(collectible)) {
			collectible.collected = true;
			var coinValue = collectible.value || 1;
			levelCoinsCollected += coinValue;
			storage.totalCoins += coinValue;
			storage.totalScore += coinValue * 10;
			LK.getSound('coinCollect').play();
			var coinDisplay = game.findChildByName || null;
			var guiChildren = LK.gui.top.children;
			for (var i = 0; i < guiChildren.length; i++) {
				if (guiChildren[i].name === 'coinDisplay') {
					guiChildren[i].setText('Coins: ' + levelCoinsCollected);
				}
			}
			collectible.destroy();
			collectibles.splice(c, 1);
			continue;
		}
	}
	// Update obstacles
	for (var o = obstacles.length - 1; o >= 0; o--) {
		var obstacle = obstacles[o];
		obstacle.lastY = obstacle.y;
		obstacle.update();
		// Check if off-screen
		if (obstacle.y > 2732) {
			obstacle.destroy();
			obstacles.splice(o, 1);
			continue;
		}
		// Check collision with player
		if (player.intersects(obstacle)) {
			gameActive = false;
			LK.showGameOver();
		}
	}
	// Check if player reached finish line
	if (finishLine && !finishLine.lastWasIntersecting && player.intersects(finishLine)) {
		finishLine.reached = true;
		finishLine.lastWasIntersecting = true;
		gameActive = false;
		storage.levelsCompleted = Math.max(storage.levelsCompleted, currentLevelData.number);
		if (currentLevelData.number < 25) {
			storage.levelsCompleted = currentLevelData.number + 1;
		}
		LK.getSound('levelComplete').play();
		showLevelCompleteScreen();
	}
	if (finishLine && player && !player.intersects(finishLine)) {
		finishLine.lastWasIntersecting = false;
	}
	// Check if player fell off
	if (player.y > 2732) {
		gameActive = false;
		LK.showGameOver();
	}
};
// Touch/drag controls
var dragNode = null;
game.move = function (x, y, obj) {
	if (player && gameState === 'gameplay') {
		player.x = x;
		player.y = y;
	}
};
game.down = function (x, y, obj) {
	if (gameState === 'gameplay' && player) {
		dragNode = player;
		player.x = x;
		player.y = y;
	}
};
game.up = function (x, y, obj) {
	dragNode = null;
};
